/* ============================================================

/* ============================================================
   SECTION 14 - AFTER CHECKOUT INVENTORY CHECK
   ============================================================

   Product 1 quantity = 1
   Product 2 quantity = 1
   Product 3 quantity = 1

   Then inventory should decrease by 1.

   Example:

   Before:
   25, 20, 30

   After:
   24, 19, 29

   ============================================================ */

    i.available_stock
FROM inventory i
JOIN products p
    ON p.id = i.product_id

   ============================================================ */

SELECT

/* ============================================================
   SECTION 21 - CHECK FOR PRODUCTS WITHOUT INVENTORY
   ============================================================

   VERY IMPORTANT.

   Every active product that can be purchased should have
   an inventory record.

   ============================================================ */

SELECT
    p.id,
    p.name,
    p.sku,
    p.stock
FROM products p
LEFT JOIN inventory i
    ON i.product_id = p.id
WHERE p.active = TRUE
  AND i.id IS NULL;


/* ============================================================
   SECTION 22 - CHECK FOR DUPLICATE INVENTORY
   ============================================================ */

SELECT
    product_id,
    COUNT(*) AS inventory_count
FROM inventory
GROUP BY product_id
HAVING COUNT(*) > 1;


/* ============================================================
   SECTION 23 - CHECK CART TOTAL CALCULATION
   ============================================================

   This helps verify:

       quantity × price = total_price

   ============================================================ */

SELECT
    ci.id,
    ci.cart_id,
    ci.product_id,
    p.name AS product_name,
    ci.quantity,
    ci.price,
    ci.total_price,

    (ci.quantity * ci.price) AS calculated_total

FROM cart_items ci

JOIN products p
    ON p.id = ci.product_id;


/* ============================================================
   SECTION 24 - CHECK ORDER TOTAL CALCULATION
   ============================================================ */

SELECT
    o.id AS order_id,
    o.total_amount AS order_saved_total,

    SUM(oi.total_price) AS calculated_total,

    o.total_items AS order_saved_items,

    SUM(oi.quantity) AS calculated_items

FROM orders o

JOIN order_items oi
    ON oi.order_id = o.id

GROUP BY
    o.id,
    o.total_amount,
    o.total_items;


/* ============================================================
   SECTION 25 - CHECK FOR DATA INCONSISTENCY
   ============================================================

   Product stock and inventory stock are separate values in
   the current database design.

   This query helps you see both.

   ============================================================ */

SELECT
    p.id AS product_id,
    p.name,
    p.stock AS product_stock,
    i.available_stock AS inventory_stock
FROM products p
LEFT JOIN inventory i
    ON i.product_id = p.id
ORDER BY p.id;
